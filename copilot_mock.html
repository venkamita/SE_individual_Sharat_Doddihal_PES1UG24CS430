1// File upload handler for Automated Rubric Assignment Evaluator
2const multer = require('multer');
3const path = require('path');
4
5const storage = multer.diskStorage({
6    destination: (req, file, cb) => {
7        cb(null, 'uploads/');
8    },
9    filename: (req, file, cb) => {
10        const uniqueName = `${Date.now()}-${file.originalname}`;
11        cb(null, uniqueName);
12    }
13});
14
15const fileFilter = (req, file, cb) => {
16    const allowedTypes = ['.zip', '.pdf', '.py', '.js'];
17    const ext = path.extname(file.originalname).toLowerCase();
18    if (allowedTypes.includes(ext)) {
19        cb(null, true);
20    } else {
21        cb(new Error('Invalid file type'), false);
22    }
23};
24
25const upload = multer({ storage, fileFilter, limits: { fileSize: 50 * 1024 * 1024 } });
26module.exports = upload;
